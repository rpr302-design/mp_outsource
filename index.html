`; }); } // विभाग सूची if (data.departments && data.departments.length > 0) { const deptContainer = document.querySelector('.dept-grid'); if (deptContainer) { deptContainer.innerHTML = data.departments.map((d, i) => `${i + 1}. ${d}`).join('
\n'); } } // लिंक्स if (data.importantLinks && data.importantLinks.length > 0) { const linksContainer = document.getElementById('dyn-links-container'); if (linksContainer) { linksContainer.innerHTML = ''; data.importantLinks.forEach(l => { linksContainer.innerHTML += `${l.title}`; }); } } // वीडियो if (data.videos && data.videos.length > 0) { const vContainer = document.getElementById('dyn-videos-container'); if (vContainer) { vContainer.innerHTML = ''; data.videos.forEach(v => { vContainer.innerHTML += `
${v.title}
`; }); } } // गैलरी if (data.gallery && data.gallery.length > 0) { const gContainer = document.getElementById('dyn-gallery-container'); if (gContainer) { gContainer.innerHTML = ''; data.gallery.forEach(img => { gContainer.innerHTML += `

`; }); } } }) .catch(err => console.log("JSON Data loading fallback:", err));
